// Forum data storage using localStorage (persists forever) class Forum { constructor() { this.posts = this.loadPosts(); this.currentPage = 1; this.postsPerPage = 10; this.currentCategory = 'all'; this.searchTerm = ''; this.sortBy = 'latest'; this.init(); } loadPosts() { const stored = localStorage.getItem('forum_posts'); if (stored) { return JSON.parse(stored); } // Sample posts to get started return [ { id: Date.now() + 1, title: "Welcome to the Community Forum! 🎉", content: "Welcome everyone! This is your space to discuss ideas, ask questions, and connect with others. Feel free to introduce yourself and start conversations!", author: "Admin", category: "announcements", date: new Date(Date.now() - 86400000).toISOString(), replies: [ { author: "Community Member", content: "Excited to be here! This is awesome!", date: new Date(Date.now() - 43200000).toISOString() } ] }, { id: Date.now() + 2, title: "What would you like to see on this website?", content: "Share your ideas and suggestions for new features or content you'd love to see!", author: "Site Creator", category: "ideas", date: new Date(Date.now() - 172800000).toISOString(), replies: [] } ]; } savePosts() { localStorage.setItem('forum_posts', JSON.stringify(this.posts)); } addPost(post) { const newPost = { id: Date.now(), ...post, date: new Date().toISOString(), replies: [] }; this.posts.unshift(newPost); this.savePosts(); return newPost; } addReply(postId, reply) { const post = this.posts.find(p => p.id === postId); if (post) { post.replies.push({ ...reply, date: new Date().toISOString() }); this.savePosts(); return true; } return false; } getFilteredPosts() { let filtered = [...this.posts]; // Filter by category if (this.currentCategory !== 'all') { filtered = filtered.filter(p => p.category === this.currentCategory); } // Filter by search if (this.searchTerm) { const term = this.searchTerm.toLowerCase(); filtered = filtered.filter(p => p.title.toLowerCase().includes(term) || p.content.toLowerCase().includes(term) || p.author.toLowerCase().includes(term) ); } // Sort switch(this.sortBy) { case 'latest': filtered.sort((a, b) => new Date(b.date) - new Date(a.date)); break; case 'oldest': filtered.sort((a, b) => new Date(a.date) - new Date(b.date)); break; case 'popular': filtered.sort((a, b) => b.replies.length - a.replies.length); break; } return filtered; } getPaginatedPosts() { const filtered = this.getFilteredPosts(); const start = (this.currentPage - 1) * this.postsPerPage; const end = start + this.postsPerPage; return { posts: filtered.slice(start, end), totalPages: Math.ceil(filtered.length / this.postsPerPage) }; } init() { this.renderPosts(); this.setupEventListeners(); } renderPosts() { const container = document.getElementById('postsContainer'); const { posts, totalPages } = this.getPaginatedPosts(); if (posts.length === 0) { container.innerHTML = `

No posts yet

Be the first to start a discussion!

`; return; } container.innerHTML = posts.map(post => `

${this.escapeHtml(post.title)} ${this.getCategoryName(post.category)}

${this.escapeHtml(post.author)} ${this.formatDate(post.date)}
${this.escapeHtml(post.content.substring(0, 150))}${post.content.length > 150 ? '...' : ''}
${post.replies.length} replies
`).join(''); // Add click handlers document.querySelectorAll('.post-card').forEach(card => { card.addEventListener('click', () => { const postId = parseInt(card.dataset.postId); this.showPostDetail(postId); }); }); this.renderPagination(totalPages); } renderPagination(totalPages) { const paginationDiv = document.getElementById('pagination'); if (totalPages <= 1) { paginationDiv.innerHTML = ''; return; } let html = ''; for (let i = 1; i <= totalPages; i++) { html += ``; } paginationDiv.innerHTML = html; document.querySelectorAll('.page-btn').forEach(btn => { btn.addEventListener('click', (e) => { this.currentPage = parseInt(btn.dataset.page); this.renderPosts(); window.scrollTo({ top: 0, behavior: 'smooth' }); }); }); } showPostDetail(postId) { const post = this.posts.find(p => p.id === postId); if (!post) return; const modal = document.getElementById('replyModal'); const replyToPost = document.getElementById('replyToPost'); replyToPost.innerHTML = `

${this.escapeHtml(post.title)}

Posted by ${this.escapeHtml(post.author)} on ${this.formatDate(post.date)}
${this.escapeHtml(post.content)}
${post.replies.length > 0 ? `
Replies (${post.replies.length}): ${post.replies.map(reply => `
${this.escapeHtml(reply.author)}
${this.formatDate(reply.date)}
${this.escapeHtml(reply.content)}
`).join('')}
` : '

No replies yet. Be the first to respond!

'}
`; modal.style.display = 'block'; // Store current post ID for reply window.currentReplyPostId = postId; } setupEventListeners() { // New post button document.getElementById('newPostBtn').addEventListener('click', () => { document.getElementById('postModal').style.display = 'block'; document.getElementById('modalTitle').innerHTML = ' Create New Post'; document.getElementById('postForm').reset(); // Load saved author name const savedAuthor = localStorage.getItem('forum_author'); if (savedAuthor) { document.getElementById('postAuthor').value = savedAuthor; document.getElementById('saveAuthor').checked = true; } }); // Close modals document.querySelector('.close').addEventListener('click', () => { document.getElementById('postModal').style.display = 'none'; }); document.querySelector('.close-reply').addEventListener('click', () => { document.getElementById('replyModal').style.display = 'none'; }); document.getElementById('cancelBtn').addEventListener('click', () => { document.getElementById('postModal').style.display = 'none'; }); document.getElementById('cancelReplyBtn').addEventListener('click', () => { document.getElementById('replyModal').style.display = 'none'; }); // Click outside modal to close window.addEventListener('click', (e) => { const postModal = document.getElementById('postModal'); const replyModal = document.getElementById('replyModal'); if (e.target === postModal) postModal.style.display = 'none'; if (e.target === replyModal) replyModal.style.display = 'none'; }); // Submit new post document.getElementById('postForm').addEventListener('submit', (e) => { e.preventDefault(); const author = document.getElementById('postAuthor').value; if (document.getElementById('saveAuthor').checked) { localStorage.setItem('forum_author', author); } this.addPost({ title: document.getElementById('postTitle').value, category: document.getElementById('postCategory').value, author: author, content: document.getElementById('postContent').value }); document.getElementById('postModal').style.display = 'none'; this.currentPage = 1; this.renderPosts(); }); // Submit reply document.getElementById('replyForm').addEventListener('submit', (e) => { e.preventDefault(); this.addReply(window.currentReplyPostId, { author: document.getElementById('replyAuthor').value, content: document.getElementById('replyContent').value }); document.getElementById('replyModal').style.display = 'none'; document.getElementById('replyForm').reset(); this.renderPosts(); }); // Category filters document.querySelectorAll('.category-btn').forEach(btn => { btn.addEventListener('click', () => { document.querySelectorAll('.category-btn').forEach(b => b.classList.remove('active')); btn.classList.add('active'); this.currentCategory = btn.dataset.category; this.currentPage = 1; this.renderPosts(); }); }); // Search document.getElementById('searchInput').addEventListener('input', (e) => { this.searchTerm = e.target.value; this.currentPage = 1; this.renderPosts(); }); // Sort document.getElementById('sortSelect').addEventListener('change', (e) => { this.sortBy = e.target.value; this.currentPage = 1; this.renderPosts(); }); } getCategoryName(category) { const names = { 'general': 'General', 'help': 'Help', 'ideas': 'Ideas', 'announcements': 'Announcements' }; return names[category] || category; } formatDate(dateString) { const date = new Date(dateString); const now = new Date(); const diffTime = Math.abs(now - date); const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24)); if (diffDays === 0) { return 'Today'; } else if (diffDays === 1) { return 'Yesterday'; } else if (diffDays < 7) { return `${diffDays} days ago`; } else { return date.toLocaleDateString(); } } escapeHtml(text) { const div = document.createElement('div'); div.textContent = text; return div.innerHTML; } } // Initialize forum when page loads document.addEventListener('DOMContentLoaded', () => { new Forum(); });
